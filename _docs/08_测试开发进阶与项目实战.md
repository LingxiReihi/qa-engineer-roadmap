### 4.4 测试数据管理

**策略**：
- 小量数据：硬编码 + fixture
- 中等数据：YAML/CSV/Excel
- 大量数据：数据库 + 工厂模式
- 动态数据：Faker

```python
from dataclasses import dataclass
from faker import Faker

@dataclass
class UserFactory:
    _fake = Faker()
    name: str = None
    email: str = None
    age: int = None

    def __post_init__(self):
        if self.name is None:
            self.name = self._fake.name()
        if self.email is None:
            self.email = self._fake.email()
        if self.age is None:
            self.age = self._fake.random_int(min=18, max=60)

    def create(self) -> dict:
        return {
            "name": self.name,
            "email": self.email,
            "age": self.age,
        }

    @classmethod
    def create_many(cls, n: int) -> list[dict]:
        return [cls().create() for _ in range(n)]
```

### T8-2：核心模块实现

**任务 A**：HTTP 客户端封装

```python
# core/client.py
import requests
from functools import lru_cache
import logging

logger = logging.getLogger(__name__)

class ApiClient:
    def __init__(self, base_url: str, timeout: int = 10):
        self.base_url = base_url.rstrip("/")
        self.timeout = timeout
        self.session = requests.Session()

    def _url(self, path: str) -> str:
        return f"{self.base_url}{path}"

    def _request(self, method: str, path: str, **kwargs) -> requests.Response:
        url = self._url(path)
        kwargs.setdefault("timeout", self.timeout)
        logger.info(f"{method} {url}")
        try:
            resp = self.session.request(method, url, **kwargs)
            logger.debug(f"Response: {resp.status_code}")
            return resp
        except requests.RequestException as e:
            logger.error(f"Request failed: {e}")
            raise

    def get(self, path: str, **kwargs) -> requests.Response:
        return self._request("GET", path, **kwargs)

    def post(self, path: str, **kwargs) -> requests.Response:
        return self._request("POST", path, **kwargs)

    def put(self, path: str, **kwargs) -> requests.Response:
        return self._request("PUT", path, **kwargs)

    def patch(self, path: str, **kwargs) -> requests.Response:
        return self._request("PATCH", path, **kwargs)

    def delete(self, path: str, **kwargs) -> requests.Response:
        return self._request("DELETE", path, **kwargs)

    def auth(self, token: str):
        self.session.headers.update({"Authorization": f"Bearer {token}"})
```

```python
import pytest
from .client import ApiClient

class BaseTest:
    """测试基类，提供常用工具方法。"""

    @pytest.fixture(autouse=True)

    def assert_success(self, response, expected_status: int = 200):
        assert response.status_code == expected_status, (
            f"Expected {expected_status}, got {response.status_code}"
        )
        return response.json()

    def assert_error(self, response, expected_status: int = 400):
        assert response.status_code == expected_status
        return response.json()

    def assert_field(self, data, field: str, expected):
        assert field in data, f"Field '{field}' not in response"
        assert data[field] == expected, f"Expected {expected}, got {data[field]}"
```

**任务 C**：数据工厂

```python
# core/data_factory.py
from dataclasses import dataclass, field
from typing import Optional
from faker import Faker

_fake = Faker()

@dataclass
class UserFactory:
    name: Optional[str] = field(default=None, repr=False)
    email: Optional[str] = field(default=None, repr=False)
    age: Optional[int] = field(default=None, repr=False)
    active: bool = True

    def __post_init__(self):
        if self.name is None:
            self.name = _fake.name()
        if self.email is None:
            self.email = _fake.email()
        if self.age is None:
            self.age = _fake.random_int(min=18, max=60)

    def to_dict(self) -> dict:
        return {
            "name": self.name,
            "email": self.email,
            "age": self.age,
            "active": self.active,
        }

    @classmethod
    def create(cls, **kwargs) -> "UserFactory":
        return cls(**kwargs)

    @classmethod
    def create_many(cls, n: int, **kwargs) -> list["UserFactory"]:
        return [cls(**kwargs) for _ in range(n)]

    @classmethod
    def create_valid(cls) -> "UserFactory":
        return cls(
            name="Alice",
            email="alice@example.com",
            age=30,
        )

    @classmethod
    def create_invalid(cls, **kwargs) -> "UserFactory":
        base = {"name": "Alice", "email": "alice@example.com", "age": 30}
        base.update(kwargs)
        return cls(**base)
```

```python
import time

def pytest_runtest_setup(item):
    item._start_time = time.perf_counter()

def pytest_runtest_teardown(item):
    elapsed = time.perf_counter() - item._start_time
    if elapsed > 5.0:
        print(f"\n⚠️  SLOW TEST: {item.name} took {elapsed:.2f}s")
```
